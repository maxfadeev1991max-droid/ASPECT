/* * Aspect Copilot Gateway * * Запуск: * node aspect-copilot-gateway.js * * Переменные окружения для серверной модели: * AI_GATEWAY_URL=https://.../v1/chat/completions * AI_GATEWAY_KEY=... * AI_MODEL=... * * Ключ никогда не попадает в HTML и не отправляется браузеру. * Без этих переменных сервер работает в прозрачном локальном fallback-режиме. */ const http = require('node:http'); const fs = require('node:fs'); const path = require('node:path'); const PORT = Number(process.env.PORT || 3000); const MAX_BODY = 320 * 1024; const HTML_FILE = path.join(__dirname, 'attached_assets', 'AspectRu_ready_1789343323309.html'); const MODEL_URL = process.env.AI_GATEWAY_URL || (process.env.OPENAI_BASE_URL ? process.env.OPENAI_BASE_URL.replace(/\/$/, '') + '/chat/completions' : ''); const MODEL_KEY = process.env.AI_GATEWAY_KEY || process.env.OPENAI_API_KEY || ''; const MODEL_NAME = process.env.AI_MODEL || ''; function send(res, status, body, type = 'application/json; charset=utf-8') { res.writeHead(status, { 'Content-Type': type, 'Cache-Control': 'no-store', 'Access-Control-Allow-Origin': process.env.CORS_ORIGIN || '*', 'Access-Control-Allow-Headers': 'Content-Type' }); res.end(type.startsWith('application/json') ? JSON.stringify(body) : body); } function parseBody(req) { return new Promise((resolve, reject) => { let raw = ''; req.setEncoding('utf8'); req.on('data', chunk => { raw += chunk; if (Buffer.byteLength(raw) > MAX_BODY) { reject(new Error('request_too_large')); req.destroy(); } }); req.on('end', () => { try { resolve(JSON.parse(raw || '{}')); } catch (error) { reject(new Error('invalid_json')); } }); req.on('error', reject); }); } function clean(value, limit = 800) { return String(value == null ? '' : value).replace(/\s+/g, ' ').trim().slice(0, limit); } function localFallback(input) { const scan = input.scan || {}; const knowledge = input.knowledge || {}; const lots = Array.isArray(knowledge.suspiciousLots) ? knowledge.suspiciousLots : []; const lines = Array.isArray(knowledge.lineSignals) ? knowledge.lineSignals : []; const defects = Array.isArray(knowledge.recurringDefects) ? knowledge.recurringDefects : []; const currentLot = clean(scan.baseLot).toLowerCase(); const currentLine = clean(scan.line).toLowerCase(); const lot = lots.find(item => currentLot && clean(item[0]).toLowerCase() === currentLot); const line = lines.find(item => currentLine && clean(item[0]).toLowerCase() === currentLine); let answer; if (lot) { answer = `Наиболее вероятная причина — дрейф лота сырья «${clean(lot[0])}». Он встречается в истории ${lot[1]} раз и совпадает с текущим лотом. Проверьте контрольный замес на другом лоте или силу пигмента до новой коррекции.`; } else if (line) { answer = `Наиболее вероятная причина — паттерн линии «${clean(line[0])}». В истории найдено ${line[1]} связанных записей. Сначала проверьте секцию печати, ракель и скорость, затем сравните результат с другой линией.`; } else if (defects[0]) { answer = `В истории повторяется сигнал «${clean(defects[0][0])}» (${defects[0][1]} раз). Проверьте карту рулона и сопоставьте период дефекта с валом, клише или секцией линии.`; } else if (!scan.dE || !scan.baseLot) { answer = 'Первопричину пока нельзя надёжно ранжировать: не хватает ΔE или лота основы. Сначала сохраните пару образцов, замер и трассировку сырья.'; } else { answer = `Повторяемой причины пока не найдено. Текущий риск запуска: ${clean(scan.risk) || '—'}/100. Сохраните следующий замес с артикулом, линией, лотом, фото и ΔE.`; } return { answer, source: 'local', model: null }; } function buildPrompt(input) { const question = clean(input.question || 'Определи вероятную первопричину и следующий шаг.', 1200); const scan = JSON.stringify(input.scan || {}).slice(0, 6000); const knowledge = JSON.stringify(input.knowledge || {}).slice(0, 18000); return [ 'Ты Aspect Copilot для производства обоев и печатных покрытий.', 'Отвечай на русском, конкретно и без выдуманных фактов.', 'Разделяй наблюдение, гипотезу и действие для проверки.', 'Не утверждай, что причина доказана, если есть только корреляция.', 'Не предлагай опасные действия на работающей линии.', `Вопрос оператора: ${question}`, `Текущее состояние: ${scan}`, `Локальная карта знаний: ${knowledge}`, 'Формат: 3–6 коротких абзацев, сначала вывод, затем доказательства и следующий безопасный шаг.' ].join('\n\n'); } async function askModel(input) { if (!MODEL_URL || !MODEL_KEY || !MODEL_NAME || typeof fetch !== 'function') return null; const upstream = await fetch(MODEL_URL, { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${MODEL_KEY}` }, body: JSON.stringify({ model: MODEL_NAME, temperature: 0.15, max_tokens: 700, messages: [ { role: 'system', content: 'Ты объяснимый производственный AI-ассистент Aspect Copilot.' }, { role: 'user', content: buildPrompt(input) } ] }) }); if (!upstream.ok) throw new Error(`upstream_${upstream.status}`); const data = await upstream.json(); const answer = data?.choices?.[0]?.message?.content || data?.output_text || data?.output?.[0]?.content?.[0]?.text; if (!answer) throw new Error('model_empty_response'); return { answer: clean(answer, 6000), source: 'model', model: MODEL_NAME }; } async function analyze(req, res) { let input; try { input = await parseBody(req); } catch (error) { return send(res, error.message === 'request_too_large' ? 413 : 400, { error: error.message }); } if (!input || typeof input !== 'object') return send(res, 400, { error: 'invalid_payload' }); try { const modelResult = await askModel(input); return send(res, 200, modelResult || localFallback(input)); } catch (error) { console.error('[Aspect Gateway] model fallback:', error.message); return send(res, 200, { ...localFallback(input), modelError: 'upstream_unavailable' }); } } function serveApp(req, res) { if (req.url !== '/' && req.url !== '/AspectRu_ready_1789343323309.html') { return send(res, 404, { error: 'not_found' }); } fs.readFile(HTML_FILE, (error, data) => { if (error) return send(res, 500, { error: 'app_file_unavailable' }); send(res, 200, data.toString('utf8'), 'text/html; charset=utf-8'); }); } const server = http.createServer((req, res) => { if (req.method === 'OPTIONS') return send(res, 204, ''); const pathname = new URL(req.url, `http://${req.headers.host || 'localhost'}`).pathname; if (req.method === 'POST' && pathname === '/api/copilot/analyze') return analyze(req, res); if (req.method === 'GET') return serveApp(req, res); return send(res, 405, { error: 'method_not_allowed' }); }); server.listen(PORT, '0.0.0.0', () => { console.log(`[Aspect Gateway] listening on http://0.0.0.0:${PORT}`); console.log(`[Aspect Gateway] model mode: ${MODEL_URL && MODEL_KEY && MODEL_NAME ? 'server model' : 'local fallback'}`); });
